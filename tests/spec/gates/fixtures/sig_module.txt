A module for the signatures gate's test, kept as .txt so no linter rewrites
it: what it exports is the point.

/** Right. */
export const derive = (salt: u8[] | null, ikm: Secret<u8[]>, n: u32): u8[] => new Uint8Array(0)

/** A wrong return, over lines. */
export const check = (
  a: string,
  b: Map<string, i32>,
): boolean => true

/** Generic: left out. */
export const pick = <T>(a: T): T => a

/** Right: words in, words or null out. */
export const quarter = (state: u32[], a: i32): u32[] | null => null

/** Declared bytes, exports u32 words. */
export const wordsOut = (k: u8[]): u32[] => []

/** Declared u32 words, exports bytes. */
export const bytesOut = (k: u32[]): u8[] => new Uint8Array(0)

/** Declared u64 words, exports u16 words. */
export const narrow = (k: u64[]): u16[] => []

/** Declared u32 words, exports u32 words or null. */
export const maybeOut = (k: u32[]): u32[] | null => null

/** Declared u32 words or null, exports u32 words. */
export const maybeIn = (k: u32[] | null): u32[] => []

/** Declared u32 words, exports one u32. */
export const oneWord = (k: u32[]): u32 => 0

/** Declared u32 words, exports u16 words. */
export const halfWords = (k: u16[]): u32[] => []

/** Right: a window, (buf, off, len). */
export const winRight = (buf: u8[], off: i32, len: i32): u32 => 0

/** Right: two windows around a plain parameter. */
export const winTwo = (a: u8[], ao: i32, al: i32, n: u32, b: u8[], bo: i32, bl: i32): u32 => 0

/** Declared a window, exports plain bytes. */
export const winPlain = (k: u8[]): u32 => 0

/** Declared a window, exports an unsigned offset and length. */
export const winUnsigned = (buf: u8[], off: u32, len: u32): u32 => 0

/** Declared a window, exports the offset and length first. */
export const winOrder = (off: i32, len: i32, buf: u8[]): u32 => 0

/** Declared a window, exports only an offset. */
export const winShort = (buf: u8[], off: i32): u32 => 0

/** Declared plain bytes, exports a window. */
export const winUndeclared = (buf: u8[], off: i32, len: i32): u32 => 0

/** Not a function. */
export const LIMIT: i32 = 3
